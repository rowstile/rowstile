-- =====================================================================
-- app_schema.sql — an example app that already exists, with its data.
-- Nothing here knows about permissions: the policy file maps onto it.
-- =====================================================================
\set ON_ERROR_STOP on
SET client_min_messages = warning;

DROP SCHEMA IF EXISTS authz_gen, authz_int, authz, test, bench, app CASCADE;
DO $$ BEGIN
  IF NOT EXISTS (SELECT FROM pg_roles WHERE rolname = 'app_user') THEN
    CREATE ROLE app_user NOLOGIN;   -- the role your application connects as
  END IF;
END $$;

CREATE SCHEMA app;

CREATE TABLE app.users (
  id   bigint PRIMARY KEY,
  name text NOT NULL
);
CREATE TABLE app.orgs (
  id   bigint PRIMARY KEY,
  name text NOT NULL
);
CREATE TABLE app.org_members (
  org_id  bigint NOT NULL REFERENCES app.orgs,
  user_id bigint NOT NULL REFERENCES app.users,
  role    text   NOT NULL CHECK (role IN ('admin', 'member')),
  PRIMARY KEY (org_id, user_id)
);
CREATE INDEX ON app.org_members (user_id);
CREATE TABLE app.teams (
  id        bigint PRIMARY KEY,
  org_id    bigint NOT NULL REFERENCES app.orgs,
  parent_id bigint REFERENCES app.teams,       -- sub-teams
  name      text   NOT NULL
);
CREATE INDEX ON app.teams (parent_id);
CREATE TABLE app.team_members (
  team_id bigint NOT NULL REFERENCES app.teams,
  user_id bigint NOT NULL REFERENCES app.users,
  PRIMARY KEY (team_id, user_id)
);
CREATE INDEX ON app.team_members (user_id);
CREATE TABLE app.folders (
  id        bigint  GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  org_id    bigint  NOT NULL REFERENCES app.orgs,
  parent_id bigint  REFERENCES app.folders ON UPDATE CASCADE,   -- delete contents first
  owner_id  bigint  REFERENCES app.users,
  name      text    NOT NULL,
  inherit   boolean NOT NULL DEFAULT true
);
CREATE INDEX ON app.folders (parent_id);
CREATE INDEX ON app.folders (owner_id);
CREATE INDEX ON app.folders (org_id);
CREATE TABLE app.files (
  id           bigint  GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  folder_id    bigint  NOT NULL REFERENCES app.folders ON UPDATE CASCADE,
  owner_id     bigint  REFERENCES app.users,
  name         text    NOT NULL,
  confidential boolean NOT NULL DEFAULT false,
  body         text
);
CREATE INDEX ON app.files (folder_id);
CREATE INDEX ON app.files (owner_id);
-- the app already lets admins give a whole team access to a folder
CREATE TABLE app.folder_team_access (
  folder_id bigint NOT NULL REFERENCES app.folders ON DELETE CASCADE ON UPDATE CASCADE,
  team_id   bigint NOT NULL REFERENCES app.teams,
  access    text   NOT NULL CHECK (access IN ('view', 'edit')),
  PRIMARY KEY (folder_id, team_id)
);
CREATE INDEX ON app.folder_team_access (team_id);
-- a folder can also appear inside other folders (like a shortcut)
CREATE TABLE app.folder_links (
  folder_id bigint NOT NULL REFERENCES app.folders ON DELETE CASCADE ON UPDATE CASCADE,
  parent_id bigint NOT NULL REFERENCES app.folders ON DELETE CASCADE ON UPDATE CASCADE,
  PRIMARY KEY (folder_id, parent_id)
);
CREATE INDEX ON app.folder_links (parent_id);

GRANT USAGE ON SCHEMA app TO app_user;
GRANT SELECT ON ALL TABLES IN SCHEMA app TO app_user;
GRANT INSERT, UPDATE, DELETE ON app.folders, app.files TO app_user;
GRANT USAGE ON ALL SEQUENCES IN SCHEMA app TO app_user;
